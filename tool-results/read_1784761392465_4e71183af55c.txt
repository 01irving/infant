   124→function formatEdad(meses: number): string {
   125→  if (meses < 1) return "< 1 mes";
   126→  if (meses === 1) return "1 mes";
   127→  if (meses < 12) return `${meses} meses`;
   128→  const a = Math.floor(meses / 12),
   129→    m = meses % 12;
   130→  return m > 0 ? `${a} a\u00f1os, ${m} meses` : `${a} a\u00f1os`;
   131→}
   132→
   133→function formatFecha(fecha: string): string {
   134→  try {
   135→    return new Date(fecha).toLocaleDateString("es-ES", {
   136→      day: "2-digit",
   137→      month: "2-digit",
   138→      year: "numeric",
   139→      hour: "2-digit",
   140→      minute: "2-digit",
   141→    });
   142→  } catch {
   143→    return fecha;
   144→  }
   145→}

   331→      setLoading(false);
   332→    }

   375→        title: "Error",
