   124→   124→function formatEdad(meses: number): string {
   125→   125→  if (meses < 1) return "< 1 mes";
   126→   126→  if (meses === 1) return "1 mes";
   127→   127→  if (meses < 12) return `${meses} meses`;
   128→   128→  const a = Math.floor(meses / 12),
   129→   129→    m = meses % 12;
   130→   130→  return m > 0 ? `${a} a\u00f1os, ${m} meses` : `${a} a\u00f1os`;
   131→   131→}
   132→   132→
   133→   133→function formatFecha(fecha: string): string {
   134→   134→  try {
   135→   135→    return new Date(fecha).toLocaleDateString("es-ES", {
   136→   136→      day: "2-digit",
   137→   137→      month: "2-digit",
   138→   138→      year: "numeric",
   139→   139→      hour: "2-digit",
   140→   140→      minute: "2-digit",
   141→   141→    });
   142→   142→  } catch {
   143→   143→    return fecha;
   144→   144→  }
   145→   145→}
