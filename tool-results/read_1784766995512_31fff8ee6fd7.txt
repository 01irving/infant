   132→def costo_crecimiento(edad_meses):
   133→    if edad_meses <= 3: return 175
   134→    if edad_meses <= 6: return 60
   135→    if edad_meses <= 12: return 20
   136→    if edad_meses <= 36: return 3
   137→    return 0
   138→
   139→def calcular_bmr(peso, edad_meses, sexo):
   140→    """Ecuaciones de Schofield por rango de edad y sexo"""
   141→    if edad_meses < 36:
   142→        if sexo == "masculino":
   143→            return 59.5 * peso - 30.4, "Schofield (<3 años, M): 59.5×peso - 30.4"
   144→        return 58.3 * peso - 31.1, "Schofield (<3 años, F): 58.3×peso - 31.1"
   145→    elif edad_meses < 120:
   146→        if sexo == "masculino":
   147→            return 22.7 * peso + 504.3, "Schofield (3-10 años, M): 22.7×peso + 504.3"
   148→        return 20.3 * peso + 485.9, "Schofield (3-10 años, F): 20.3×peso + 485.9"
   149→    else:
   150→        if sexo == "masculino":
   151→            return 17.7 * peso + 658.2, "Schofield (10-18 años, M): 17.7×peso + 658.2"
   152→        return 13.4 * peso + 692.6, "Schofield (10-18 años, F): 13.4×peso + 692.6"
   153→
   154→def calcular_tee(peso, edad_meses, sexo):
   155→    """TEE - Método DLW (Doubly Labeled Water)"""
   156→    if edad_meses <= 24:
   157→        return -99.4 + 88.6 * peso, "TEE infantes (DLW): -99.4 + 88.6×peso"
   158→    w2 = peso * peso
   159→    if sexo == "masculino":
   160→        return 310.2 + 63.3 * peso - 0.263 * w2, "TEE niños (DLW+HR): 310.2 + 63.3×p - 0.263×p²"
   161→    return 263.4 + 65.3 * peso - 0.454 * w2, "TEE niñas (DLW+HR): 263.4 + 65.3×p - 0.454×p²"
   162→
   163→PAL_VALUES = {
   164→    "sedentario": 1.2, "ligero": 1.5, "moderado": 1.7,
   165→    "activo": 1.9, "muy_activo": 2.0
   166→}
   167→
   168→def calcular_liquidos(peso):
   169→    """Holliday-Segar por peso"""
   170→    if peso <= 10: return peso * 100
   171→    if peso <= 20: return 1000 + (peso - 10) * 50
   172→    return 1500 + (peso - 20) * 20
   173→
   174→def calcular_proteinas(edad_meses, sexo, es_prematuro):
   175→    """EAR/RDA por rangos de edad"""
   176→    if es_prematuro:
   177→        return 3.0, 4.0, "3.0 - 4.0 g/kg/día (Prematuro - OMS/FAO/UNU)"
   178→    if edad_meses <= 6: return 1.52, 1.52, "1.52 g/kg/día (AI - leche materna)"
   179→    if edad_meses <= 12: return 1.0, 1.2, "1.0 - 1.2 g/kg/día (EAR-RDA)"
   180→    if edad_meses <= 36: return 0.87, 1.05, "0.87 - 1.05 g/kg/día (EAR-RDA)"
   181→    if edad_meses <= 156: return 0.76, 0.95, "0.76 - 0.95 g/kg/día (EAR-RDA)"
   182→    if sexo == "masculino": return 0.73, 0.85, "0.73 - 0.85 g/kg/día (EAR-RDA)"
   183→    return 0.71, 0.85, "0.71 - 0.85 g/kg/día (EAR-RDA)"
   184→
   185→def calcular_sodio(edad_gestacional, edad_meses):
   186→    if edad_gestacional is not None and edad_gestacional < 37 and edad_meses <= 1:
   187→        return 3.0, "2.0 - 5.0 mEq/kg/día (Prematuro)"
   188→    return 1.5, "1.0 - 2.0 mEq/kg/día"
   189→
   190→def calcular_potasio(edad_gestacional, edad_meses):
   191→    if edad_gestacional is not None and edad_gestacional < 37 and edad_meses <= 1:
   192→        return 1.5, "1.0 - 2.0 mEq/kg/día (Prematuro)"
   193→    return 1.5, "1.0 - 2.0 mEq/kg/día"
   194→
   195→def calcular_superficie_corporal(peso, talla_cm=None, edad_meses=None):
   196→    """Fórmula de Mosteller"""
   197→    if talla_cm is None:
   198→        if edad_meses is not None:
   199→            if edad_meses <= 6: talla_cm = 50 + edad_meses * 2.5
   200→            elif edad_meses <= 12: talla_cm = 65 + (edad_meses - 6) * 1.2
   201→            else: talla_cm = 75 + (edad_meses / 12) * 6
   202→        else:
   203→            talla_cm = 55
   204→    return math.sqrt((peso * talla_cm) / 3600)

   208→def velocidad_peso_esperada(edad_meses):
   209→    if edad_meses <= 3: return "200 g/semana (0-3 meses)"
   210→    if edad_meses <= 6: return "130 g/semana (3-6 meses)"
   211→    if edad_meses <= 9: return "85 g/semana (6-9 meses)"
   212→    if edad_meses <= 12: return "75 g/semana (9-12 meses)"
   213→    if edad_meses <= 24: return "~25 g/semana (duplica al 4m, triplica al 12m)"
   214→    return "Variable por individuo; usar curvas OMS"

   216→def velocidad_talla_esperada(edad_meses):
   217→    if edad_meses <= 12: return "25 cm en el primer año (~2 cm/mes)"
   218→    if edad_meses <= 24: return "12 cm en el segundo año (~1 cm/mes)"
   219→    if edad_meses <= 132: return "5-7 cm/año (crecimiento estable)"
   220→    return "Variable con pubertad; pico ~8-12 cm/año"

   222→def perimetro_cefalico_esperado(edad_meses, sexo):
   223→    adulto = 56.5 if sexo == "masculino" else 54.5
   224→    if edad_meses <= 12: return 34.5 + edad_meses * 1.0
   225→    if edad_meses <= 24: return 46.5 + ((edad_meses - 12) / 12) * 2
   226→    hc2 = 48.5
   227→    if edad_meses <= 216:
   228→        p = min(1, (edad_meses - 24) / (216 - 24))
   229→        return hc2 + (adulto - hc2) * p
   230→    return adulto
   231→
   232→def clasificar_perimetro_cefalico(actual, esperado):
   233→    d = ((actual - esperado) / esperado) * 100
   234→    if d < -10: return "Bajo (<-2 DE) - Evaluar microcefalia", "red"
   235→    if d < -5: return "Ligeramente bajo (-1 a -2 DE)", "yellow"
   236→    if d <= 5: return "Normal (±2 DE)", "green"
   237→    if d <= 10: return "Ligeramente alto (+1 a +2 DE)", "yellow"
   238→    return "Alto (>+2 DE) - Evaluar macrocefalia", "yellow"
   239→
   240→def calcular_bmi(peso, talla_cm):
   241→    talla_m = talla_cm / 100
   242→    return round(peso / (talla_m ** 2), 2)
   243→
   244→def clasificar_bmi(bmi, edad_meses):
   245→    if edad_meses < 24:
   246→        return f"{bmi} - Usar curvas OMS para <2 años", "yellow"
   247→    if bmi < 18.5: return f"{bmi} - Bajo peso", "orange"
   248→    if bmi < 25: return f"{bmi} - Normal", "green"
   249→    if bmi < 30: return f"{bmi} - Sobrepeso", "yellow"
   250→    return f"{bmi} - Obesidad", "red"
   251→
   252→def peso_esperado_para_talla(talla_cm, sexo):
   253→    m = sexo == "masculino"
   254→    if talla_cm <= 50: return 3.2 if m else 3.1
   255→    if talla_cm <= 60: return 5.7 if m else 5.5
   256→    if talla_cm <= 70: return 7.8 if m else 7.5
   257→    if talla_cm <= 80: return 10.2 if m else 9.8
   258→    if talla_cm <= 90: return 12.5 if m else 12.0
   259→    if talla_cm <= 100: return 15.5 if m else 15.0
   260→    if talla_cm <= 110: return 18.5 if m else 18.0
   261→    if talla_cm <= 120: return 22.0 if m else 21.5
   262→    if talla_cm <= 130: return 26.5 if m else 26.0
   263→    if talla_cm <= 140: return 32.5 if m else 32.0
   264→    if talla_cm <= 150: return 40.0 if m else 40.5
   265→    if talla_cm <= 160: return 48.5 if m else 49.5
   266→    if talla_cm <= 170: return 58.0 if m else 58.0
   267→    if talla_cm <= 180: return 67.0 if m else 65.0
   268→    return 72.0 if m else 68.0

   270→def clasificar_peso_talla(pct):
   271→    if pct > 120: return f"{pct}% - Obesidad", "red"
   272→    if pct > 110: return f"{pct}% - Sobrepeso", "yellow"
   273→    if pct >= 90: return f"{pct}% - Normal", "green"
   274→    if pct >= 80: return f"{pct}% - Desnutrición leve", "orange"
   275→    if pct >= 70: return f"{pct}% - Desnutrición moderada", "orange"
   276→    return f"{pct}% - Desnutrición severa", "red"
   277→
   278→def talla_esperada_para_edad(edad_meses, sexo):
   279→    m = sexo == "masculino"
   280→    if edad_meses <= 3: return (50 + edad_meses * 3.8) if m else (49.1 + edad_meses * 3.7)
   281→    if edad_meses <= 6: return (61.4 + (edad_meses - 3) * 2.7) if m else (60.2 + (edad_meses - 3) * 2.5)
   282→    if edad_meses <= 12: return (69.5 + (edad_meses - 6) * 2.1) if m else (67.7 + (edad_meses - 6) * 2.0)
   283→    if edad_meses <= 24: return (82.1 + (edad_meses - 12) * 1.0) if m else (79.7 + (edad_meses - 12) * 1.0)
   284→    if edad_meses <= 60: return (94.1 + (edad_meses - 24) * 0.6) if m else (91.7 + (edad_meses - 24) * 0.6)
   285→    if edad_meses <= 120: return (115.7 + (edad_meses - 60) * 0.55) if m else (114.3 + (edad_meses - 60) * 0.6)
   286→    if edad_meses <= 156: return (148.7 + (edad_meses - 120) * 0.65) if m else (149.9 + (edad_meses - 120) * 0.35)
   287→    return 172.1 if m else 162.5

   319→def evaluacion_global(pt_class, te_class, bmi_class, muac_class):
   320→    severo = sum(1 for s in [pt_class, te_class, bmi_class, muac_class] if "sever" in s.lower())
   321→    moderado = sum(1 for s in [pt_class, te_class, bmi_class, muac_class] if "moderad" in s.lower())
   322→    leve = sum(1 for s in [pt_class, te_class, bmi_class, muac_class] if "leve" in s.lower() or "riesgo" in s.lower())
   323→    if severo > 0:
   324→        return "Desnutrición severa - Intervención urgente requerida", "red"
   325→    if moderado > 0:
   326→        return "Desnutrición moderada - Requiere intervención nutricional", "orange"
   327→    if leve > 0:
   328→        return "Riesgo nutricional - Monitoreo y seguimiento recomendado", "yellow"
   329→    if "Sobrepeso" in bmi_class or "Obesidad" in bmi_class:
   330→        return "Exceso de peso - Evaluación dietética recomendada", "yellow"
   331→    return "Estado nutricional normal - Continuar monitoreo regular", "green"

   337→def calcular_todo(peso, edad_meses, sexo, edad_gestacional=None, talla=None,
   338→                   nivel_actividad="moderado", perimetro_cefalico=None,
   339→                   circunferencia_brazo=None, pliegue_cutaneo=None):
   340→    pal = PAL_VALUES.get(nivel_actividad, 1.7)
   341→    es_prematuro = edad_gestacional is not None and edad_gestacional < 37 and edad_meses <= 1
   342→
   343→    bmr, metodo_bmr = calcular_bmr(peso, edad_meses, sexo)
   344→    tee, metodo_tee = calcular_tee(peso, edad_meses, sexo)
   345→    req_energetico = bmr * pal + costo_crecimiento(edad_meses)
   346→    crecimiento = costo_crecimiento(edad_meses)
   347→    liquidos = calcular_liquidos(peso)
   348→    p_ear, p_rda, p_rango = calcular_proteinas(edad_meses, sexo, es_prematuro)
   349→    sodio_val, sodio_rango = calcular_sodio(edad_gestacional, edad_meses)
   350→    potasio_val, potasio_rango = calcular_potasio(edad_gestacional, edad_meses)
   351→    sc = calcular_superficie_corporal(peso, talla, edad_meses)
   352→
   353→    r = {
   354→        "bmr": round(bmr, 1), "metodo_bmr": metodo_bmr,
   355→        "tee": round(tee, 1), "metodo_tee": metodo_tee,
   356→        "req_energetico": round(req_energetico, 1),
   357→        "req_energetico_kg": round(req_energetico / peso, 1),
   358→        "crecimiento": round(crecimiento, 1),
   359→        "liquidos": round(liquidos, 1),
   360→        "liquidos_kg": round(liquidos / peso, 1),
   361→        "p_ear": p_ear, "p_rda": p_rda, "p_rango": p_rango,
   362→        "p_total": round(p_rda * peso, 1),
   363→        "sodio_val": sodio_val, "sodio_rango": sodio_rango,
   364→        "potasio_val": potasio_val, "potasio_rango": potasio_rango,
   365→        "sc": round(sc, 3), "pal": pal, "es_prematuro": es_prematuro,
   366→    }
   367→
   368→    # Cap. 1.2 - Antropometría
   369→    r["vel_peso"] = velocidad_peso_esperada(edad_meses)
   370→    r["vel_talla"] = velocidad_talla_esperada(edad_meses)
   371→
   372→    if talla:
   373→        bmi = calcular_bmi(peso, talla)
   374→        bmi_cl, bmi_color = clasificar_bmi(bmi, edad_meses)
   375→        pt_pct = round((peso / peso_esperado_para_talla(talla, sexo)) * 100)
   376→        pt_cl, pt_color = clasificar_peso_talla(pt_pct)
   377→        te_pct = round((talla / talla_esperada_para_edad(edad_meses, sexo)) * 100)
   378→        te_cl, te_color = clasificar_talla_edad(te_pct)
   379→
   380→        r["bmi"] = bmi
   381→        r["bmi_cl"] = bmi_cl
   382→        r["bmi_color"] = bmi_color
   383→        r["pt_pct"] = pt_pct
   384→        r["pt_cl"] = pt_cl
   385→        r["pt_color"] = pt_color
   386→        r["te_pct"] = te_pct
   387→        r["te_cl"] = te_cl
   388→        r["te_color"] = te_color
   389→
   390→    if edad_meses <= 24 and perimetro_cefalico:

   394→        r["pc_cl"] = pc_cl
   395→        r["pc_color"] = pc_color
   396→
   397→    if circunferencia_brazo:
   398→        muac_cl, muac_color = clasificar_muac(circunferencia_brazo, edad_meses)
   399→        r["muac_cl"] = muac_cl
   400→        r["muac_color"] = muac_color
   401→
   402→    if pliegue_cutaneo:
   403→        pct_cl, pct_color = evaluar_pliegue(pliegue_cutaneo, edad_meses)
   404→        r["pct_cl"] = pct_cl
   405→        r["pct_color"] = pct_color
   406→
   407→    # Evaluación global
   408→    all_clases = [r.get("pt_cl", ""), r.get("te_cl", ""), r.get("bmi_cl", ""), r.get("muac_cl", "")]
   409→    if all(c for c in all_clases):
   410→        ev, ev_color = evaluacion_global(*all_clases)
   411→        r["eval"] = ev
   412→        r["eval_color"] = ev_color
   413→
   414→    return r
